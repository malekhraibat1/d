import SwiftUI

extension Color {
    static let solarPrimary = Color(red: 1.0, green: 0.62, blue: 0.0) // Orange
    static let infoPurple = Color(red: 0.75, green: 0.47, blue: 0.94)
    static let successGreen = Color(red: 0.19, green: 0.82, blue: 0.35)
    static let tertiary = Color.gray
}

extension Font {
    static let bodyM = Font.system(size: 16, weight: .medium, design: .default)
    static let caption2 = Font.system(size: 12, weight: .regular, design: .default)
}

struct Spacing {
    static let sm: CGFloat = 8
    static let md: CGFloat = 16
}

struct Radius {
    static let sm: CGFloat = 8
    static let md: CGFloat = 12
}

struct Badge: View {
    let text: String
    let color: Color
    let icon: String
    
    var body: some View {
        HStack(spacing: 4) {
            Image(systemName: icon).font(.system(size: 10))
            Text(text).font(.system(size: 10, weight: .bold))
        }
        .padding(.horizontal, 6)
        .padding(.vertical, 2)
        .background(color.opacity(0.2))
        .foregroundStyle(color)
        .clipShape(Capsule())
    }
}

struct StatusDot: View {
    let color: Color
    var body: some View {
        Circle()
            .fill(color)
            .frame(width: 8, height: 8)
    }
}
