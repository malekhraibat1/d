import SwiftUI
import SwiftData

struct ARMapManagerView: View {
    @Environment(\.modelContext) private var ctx
    @Environment(\.dismiss) private var dismiss

    let roof: RoofPlane
    let service: ARRoofOverlayService

    @State private var showingNewMapSheet = false
    @State private var newMapName = ""
    @State private var isCapturing = false
    @State private var captureError: String?

    var maps: [PersistentARMap] {
        (roof.arMaps ?? []).sorted { $0.lastUsedAt > $1.lastUsedAt }
    }

    var body: some View {
        NavigationStack {
            List {
                if !service.mapPersistence.hasActiveMap {
                    Section {
                        Button {
                            showingNewMapSheet = true
                        } label: {
                            HStack {
                                Image(systemName: "cube.transparent.fill")
                                    .foregroundStyle(.solarPrimary)
                                VStack(alignment: .leading) {
                                    Text("حفظ الخريطة الحالية")
                                        .font(.bodyM)
                                    Text("يحتاج 30-60 ثانية من المسح")
                                        .font(.caption2)
                                        .foregroundStyle(.secondary)
                                }
                                Spacer()
                                if isCapturing {
                                    ProgressView()
                                }
                            }
                        }
                        .disabled(isCapturing)
                    } header: {
                        Text("خريطة جديدة")
                    } footer: {
                        Text("يحفظ مواقع الجدران والأثاث والإضاءة في العالم الحقيقي. عند العودة للموقع، يستعيد AR كل شيء في مكانه.")
                            .font(.caption2)
                    }
                } else {
                    Section("خريطة نشطة") {
                        if let active = service.mapPersistence.currentMap {
                            activeMapCard(active)
                        }
                        Button("إلغاء التفعيل", role: .destructive) {
                            service.mapPersistence.reset()
                        }
                    }
                }

                if !maps.isEmpty {
                    Section("الخرائط المحفوظة") {
                        ForEach(maps) { map in
                            mapRow(map)
                        }
                        .onDelete { idx in
                            idx.forEach { i in
                                service.mapPersistence.delete(maps[i], context: ctx)
                            }
                        }
                    }
                }
            }
            .navigationTitle("الخرائط الدائمة")
            .navigationBarTitleDisplayMode(.inline)
            .toolbar {
                ToolbarItem(placement: .cancellationAction) {
                    Button("إغلاق") { dismiss() }
                }
            }
            .alert("اسم الخريطة", isPresented: $showingNewMapSheet) {
                TextField("مثلاً: سطح العميل أحمد", text: $newMapName)
                Button("حفظ") {
                    Task { await capture() }
                }
                Button("إلغاء", role: .cancel) {
                    newMapName = ""
                }
            }
            .alert("خطأ", isPresented: Binding(
                get: { captureError != nil },
                set: { if !$0 { captureError = nil } }
            )) {
                Button("حسنًا") { captureError = nil }
            } message: {
                Text(captureError ?? "")
            }
        }
    }

    func activeMapCard(_ map: PersistentARMap) -> some View {
        VStack(alignment: .leading, spacing: 6) {
            HStack {
                StatusDot(color: .successGreen)
                Text(map.name).font(.bodyM)
            }
            Text(map.summary)
                .font(.caption2)
                .foregroundStyle(.secondary)
            HStack {
                Image(systemName: "checkmark.seal.fill")
                    .foregroundStyle(.successGreen)
                Text("يعمل الآن")
                    .font(.caption.bold())
                    .foregroundStyle(.successGreen)
            }
        }
        .padding(.vertical, 4)
    }

    func mapRow(_ map: PersistentARMap) -> some View {
        Button {
            Task { await restore(map) }
        } label: {
            HStack(spacing: Spacing.sm) {
                Image(systemName: "map.fill")
                    .font(.system(size: 16, weight: .semibold))
                    .foregroundStyle(.infoPurple)
                    .frame(width: 40, height: 40)
                    .background(Color.infoPurple.opacity(0.12))
                    .clipShape(RoundedRectangle(cornerRadius: Radius.sm))

                VStack(alignment: .leading, spacing: 3) {
                    Text(map.name).font(.bodyM)
                    Text(map.summary)
                        .font(.caption2)
                        .foregroundStyle(.secondary)
                    HStack(spacing: 4) {
                        Text("آخر استخدام:")
                            .font(.caption2)
                            .foregroundStyle(.tertiary)
                        Text(map.lastUsedAt.formatted(.relative(presentation: .named)))
                            .font(.caption2)
                            .foregroundStyle(.tertiary)
                    }
                }

                Spacer()

                if service.mapPersistence.currentMap?.id == map.id {
                    Image(systemName: "checkmark.circle.fill")
                        .foregroundStyle(.successGreen)
                } else {
                    Image(systemName: "arrow.right.circle")
                        .foregroundStyle(.solarPrimary)
                }
            }
        }
        .buttonStyle(.plain)
    }

    func capture() async {
        isCapturing = true
        if let saved = await service.saveCurrentMap(roof: roof, name: newMapName) {
            ctx.insert(saved)
            roof.arMaps?.append(saved)
            try? ctx.save()
            newMapName = ""
        } else {
            captureError = "لم يتمكن التطبيق من حفظ الخريطة. تأكد من استقرار التتبع."
        }
        isCapturing = false
    }

    func restore(_ map: PersistentARMap) async {
        await service.restoreMap(map)
    }
}
