import Foundation
import SwiftData
import simd

enum AutoElementType: String, Codable, CaseIterable {
    case wall         = "جدار"
    case floor        = "أرضية"
    case ceiling      = "سقف"
    case roofEdge     = "حد السطح"
    case corner       = "زاوية"
    case door         = "باب"
    case window       = "نافذة"
    case obstacle     = "عقبة"
    case ac           = "وحدة تكييف"
    case chimney      = "مدخنة"
    case waterTank    = "خزان ماء"
    case skylight     = "نافذة سقفية"
    case pipe         = "أنبوب"

    var icon: String {
        switch self {
        case .wall: return "rectangle.portrait.fill"
        case .floor: return "square.fill"
        case .ceiling: return "square.topthird.inset.filled"
        case .roofEdge: return "square.dashed"
        case .corner: return "arrow.up.left.and.arrow.down.right"
        case .door: return "door.left.hand.open"
        case .window: return "window.vertical.open"
        case .obstacle: return "exclamationmark.triangle.fill"
        case .ac: return "snowflake"
        case .chimney: return "smoke.fill"
        case .waterTank: return "drop.fill"
        case .skylight: return "rectangle.inset.filled.and.person.filled"
        case .pipe: return "capsule.fill"
        }
    }

    var colorHex: String {
        switch self {
        case .wall: return "0A84FF"
        case .floor: return "30D158"
        case .ceiling: return "BF5AF2"
        case .roofEdge: return "FF9F0A"
        case .corner: return "FFD60A"
        case .door: return "64D2FF"
        case .window: return "5AC8FA"
        case .obstacle, .ac, .chimney, .waterTank: return "FF453A"
        case .skylight: return "FF9F0A"
        case .pipe: return "8E8E93"
        }
    }

    var category: ElementCategory {
        switch self {
        case .wall, .floor, .ceiling, .roofEdge, .corner: return .structure
        case .door, .window, .skylight, .pipe: return .opening
        case .obstacle, .ac, .chimney, .waterTank: return .obstacle
        }
    }
}

enum ElementCategory: String, Codable {
    case structure = "هيكل"
    case opening   = "فتحة"
    case obstacle  = "عقبة"
}

@Model
final class AutoDetectedElement {
    var id: UUID = UUID()
    var typeRaw: String = AutoElementType.wall.rawValue

    // الموقع والأبعاد
    var centerX: Float = 0
    var centerY: Float = 0
    var centerZ: Float = 0

    var width: Double = 0
    var height: Double = 0
    var depth: Double = 0
    var area: Double = 0

    // الدوران (quaternion)
    var qx: Float = 0
    var qy: Float = 0
    var qz: Float = 0
    var qw: Float = 1

    // للمضلع (roofEdge, corner)
    var polygonData: Data = Data()    // [[Float]] of points

    // الحالة
    var isConfirmed: Bool = false      // أكّده المستخدم
    var isIgnored: Bool = false        // تجاهله
    var confidence: Double = 0.8
    var detectionMethod: String = "plane"

    var notes: String = ""
    var capturedAt: Date = Date()
    var roof: RoofPlane?

    init(type: AutoElementType = .wall,
         center: SIMD3<Float> = .zero,
         width: Double = 0, height: Double = 0) {
        self.typeRaw = type.rawValue
        self.centerX = center.x
        self.centerY = center.y
        self.centerZ = center.z
        self.width = width
        self.height = height
    }

    var type: AutoElementType {
        get { AutoElementType(rawValue: typeRaw) ?? .wall }
        set { typeRaw = newValue.rawValue }
    }

    var center: SIMD3<Float> {
        SIMD3(centerX, centerY, centerZ)
    }

    var rotation: simd_quatf {
        simd_quatf(imag: SIMD3(qx, qy, qz), real: qw)
    }

    var polygon: [SIMD3<Float>] {
        get {
            guard let decoded = try? JSONDecoder().decode(
                [[Float]].self, from: polygonData
            ) else { return [] }
            return decoded.map { SIMD3($0[0], $0[1], $0[2]) }
        }
        set {
            let array = newValue.map { [$0.x, $0.y, $0.z] }
            polygonData = (try? JSONEncoder().encode(array)) ?? Data()
        }
    }

    var displayDimension: String {
        if area > 0 {
            return String(format: "%.2f م²", area)
        }
        if width > 0 && height > 0 {
            return String(format: "%.2f × %.2f م", width, height)
        }
        if width > 0 {
            return String(format: "%.2f م", width)
        }
        return "—"
    }
}
