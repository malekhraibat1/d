import Foundation
import SwiftData
import simd

@Model
final class RoofPlane {
    var id: UUID = UUID()
    var label: String = "سطح جديد"
    var latitude: Double = 24.7
    var longitude: Double = 46.7
    
    @Relationship(deleteRule: .cascade, inverse: \AutoDetectedElement.roof)
    var autoDetectedElements: [AutoDetectedElement]? = []
    
    @Relationship(deleteRule: .cascade, inverse: \PersistentARMap.roof)
    var arMaps: [PersistentARMap]? = []
    
    @Relationship(deleteRule: .cascade, inverse: \SunPathAnchor.roof)
    var sunPathAnchors: [SunPathAnchor]? = []
    
    var drawingData: Data? = nil // لتمثيل المخطط
    
    init(label: String = "سطح جديد") {
        self.label = label
    }
}

// إضافة وهمية لـ Drawing لضمان عمل الكود
struct RoofDrawing {
    var netUsableArea: Double = 0.0
    var sketches: [String] = []
}

extension RoofPlane {
    var drawing: RoofDrawing? {
        return RoofDrawing() // محاكاة
    }
}
